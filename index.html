External Household Access & Permanent Firestore Security Rules

Paste these rules in the Firebase Console → Firestore Database → Rules tab so family members across external networks and households can read and write to media, users, socialFeed, and catalogues without 30-day test-mode expiration locks: 

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    // Replace "myemail@gmail.com" with your exact Google Admin email address
    function isAdmin() {
      return isSignedIn() && request.auth.token.email == "alex.gavin05@gmail.com";
    }

    match /users/{profileId} {
      allow read: if isSignedIn();
      allow create: if isAdmin() || (
        isSignedIn() &&
        request.resource.data.ownerUid == request.auth.uid
      );
      allow update: if isAdmin() || (
        isSignedIn() && (
          resource.data.ownerUid == request.auth.uid ||
          (!('ownerUid' in resource.data) && request.resource.data.ownerUid == request.auth.uid) ||
          (resource.data.ownerUid == "" && request.resource.data.ownerUid == request.auth.uid) ||
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['activityLog', 'notifications', 'updatedAt'])
        )
      );
      allow delete: if isAdmin();
    }

    match /media/{mediaDocId} {
      allow read: if isSignedIn();
      allow create, update: if isAdmin() || (
        isSignedIn() &&
        mediaDocId == ("lib_" + request.resource.data.profileId) &&
        get(/databases/$(database)/documents/users/$(request.resource.data.profileId)).data.ownerUid == request.auth.uid
      );
      allow delete: if isAdmin();
    }

    match /socialFeed/{feedDocId} {
      allow read: if isSignedIn();
      allow create, update: if isSignedIn();
      allow delete: if isAdmin();
    }

    match /catalogues/{catalogueId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update: if isAdmin() || (
        isSignedIn() && (
          resource.data.ownerUid == request.auth.uid ||
          (resource.data.type == 'sharedList' && request.auth.uid in resource.data.get('memberUids', [])) ||
          !('ownerUid' in resource.data)
        )
      );
      allow delete: if isAdmin() || (
        isSignedIn() && resource.data.get('ownerUid', '') == request.auth.uid
      );
    }

    match /appState/{docId} {
      allow read: if isSignedIn();
      allow write: if isSignedIn();
    }
  }
}
